cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(vfl-rife VERSION 1.0 LANGUAGES C CXX)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-info.json" "{\"compiler\":\"${CMAKE_CXX_COMPILER_ID}\",\"compilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"cmakeVersion\":\"${CMAKE_VERSION}\",\"generator\":\"${CMAKE_GENERATOR}\"}\n")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT VFL_NCNN_SOURCE OR NOT EXISTS "${VFL_NCNN_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Pass the verified pinned ncnn source as VFL_NCNN_SOURCE")
endif()
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_VERSION 20250916 CACHE STRING "Pinned source version" FORCE)
set(NCNN_SIMPLEVK ON CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(NCNN_AVX512 OFF CACHE BOOL "" FORCE)
set(NCNN_INT8 OFF CACHE BOOL "" FORCE)
set(NCNN_PIXEL_ROTATE OFF CACHE BOOL "" FORCE)
set(NCNN_PIXEL_AFFINE OFF CACHE BOOL "" FORCE)
set(NCNN_PIXEL_DRAWING OFF CACHE BOOL "" FORCE)
set(NCNN_VULKAN_ONLINE_SPIRV ON CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(SKIP_GLSLANG_INSTALL ON CACHE BOOL "" FORCE)
# Only the operations required by the hash-pinned 4.25/4.26 converted graphs.
set(VFL_LAYERS binaryop cast concat convolution crop deconvolution eltwise flatten gemm innerproduct input interp memorydata packing padding pixelshuffle relu reshape sigmoid slice split)
file(GLOB VFL_LAYER_SOURCES "${VFL_NCNN_SOURCE}/src/layer/*.cpp")
foreach(layer_source IN LISTS VFL_LAYER_SOURCES)
  get_filename_component(layer_name "${layer_source}" NAME_WE)
  if(layer_name IN_LIST VFL_LAYERS)
    set(WITH_LAYER_${layer_name} ON CACHE BOOL "" FORCE)
  else()
    set(WITH_LAYER_${layer_name} OFF CACHE BOOL "" FORCE)
  endif()
endforeach()
add_subdirectory("${VFL_NCNN_SOURCE}" "${CMAKE_BINARY_DIR}/ncnn")
set(VFL_SHADER_HEADERS)
foreach(shader warp warp_pack4)
  set(header "${CMAKE_CURRENT_BINARY_DIR}/${shader}.comp.hex.h")
  add_custom_command(OUTPUT "${header}"
    COMMAND "${CMAKE_COMMAND}" "-DSHADER_SRC=${CMAKE_CURRENT_SOURCE_DIR}/vendor/rife/${shader}.comp" "-DSHADER_COMP_HEADER=${header}" -P "${CMAKE_CURRENT_SOURCE_DIR}/vendor/rife/generate_shader_comp_header.cmake"
    DEPENDS "vendor/rife/${shader}.comp" "vendor/rife/generate_shader_comp_header.cmake" VERBATIM)
  list(APPEND VFL_SHADER_HEADERS "${header}")
endforeach()
add_executable(vfl-rife main.cpp vendor/rife/warp.cpp ${VFL_SHADER_HEADERS})
target_include_directories(vfl-rife PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" vendor/rife)
target_link_libraries(vfl-rife PRIVATE ncnn)
if(MSVC)
  target_compile_options(vfl-rife PRIVATE /W4 /EHsc /utf-8)
  target_link_options(vfl-rife PRIVATE "/MANIFEST:EMBED" "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
else()
  target_compile_options(vfl-rife PRIVATE -Wall -Wextra -Wpedantic)
  # Keep the C++ runtime portable; GPU driver and libc remain host prerequisites.
  target_link_options(vfl-rife PRIVATE -static-libgcc -static-libstdc++)
endif()
install(TARGETS vfl-rife RUNTIME DESTINATION .)
